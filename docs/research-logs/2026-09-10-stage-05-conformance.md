#!/usr/bin/env python3
"""Stage 5 document/arithmetic checks; not customer or installed-skill tests."""
from decimal import Decimal as D, ROUND_CEILING
from pathlib import Path
import hashlib
import json
import re
import sys

root = Path(sys.argv[1] if len(sys.argv) > 1 else '.')
logs = root / 'docs/research-logs'
prefix = '2026-09-10-stage-05-'
a = (logs / (prefix + 'offer-and-pricing-architecture.md')).read_text()
m = (logs / (prefix + 'money-model-taxonomy.md')).read_text()
r = (logs / (prefix + 'research-and-checks.md')).read_text()
checks = []

def check(name, ok, detail=''):
    checks.append((name, bool(ok), str(detail)))

def table_rows(text, pattern):
    return [list(map(str.strip, line.split('|')[1:-1]))
            for line in text.splitlines() if re.match(pattern, line)]

# Enumerations transcribed from original main bootstrap section 12, not
# inferred from the authored tables. The model counts are independent checks.
offers = ['customer', 'outcome', 'scope', 'mechanism', 'deliverables',
          'time-to-value', 'price', 'payment structure', 'risk reversal',
          'guarantee / refund terms', 'bonuses / added components',
          'proof / credibility', 'constraints / exclusions',
          'urgency / scarcity only where truthful']
prices = ['cost-informed boundaries', 'value-based pricing',
          'competitor/reference pricing', 'willingness-to-pay', 'tiering',
          'packaging', 'usage pricing', 'subscription', 'one-time purchase',
          'service retainers', 'performance-linked structures where lawful/appropriate',
          'freemium / free-to-paid', 'discounting', 'payment timing']
models = ['initial transaction', 'recurring revenue', 'upsell', 'cross-sell',
          'expansion', 'renewal', 'usage', 'services', 'licensing',
          'marketplace take rate', 'advertising', 'affiliate / referral revenue',
          'open-source commercialisation']
fields = ['value exchange', 'payer', 'pricing unit', 'gross revenue',
          'direct costs', 'margin', 'cash timing', 'retention dependency',
          'capacity dependency', 'risk']
for name, letter, expected in [('offer', 'O', offers), ('pricing', 'P', prices)]:
    rows = table_rows(a, r'^\| ' + letter + r'\d{2} \|')
    check(name + ': all 14 literal topics',
          [(v[0], v[1]) for v in rows] == [(f'{letter}{i:02}', x) for i, x in enumerate(expected, 1)])
    check(name + ': substantive definition and boundary',
          len(rows) == 14 and all(len(v) == 4 and all(len(c) >= 60 for c in v[2:]) for v in rows))
headers = re.findall(r'^## M(\d{2}): (.+)$', m, re.M)
check('13 literal model headings', headers == [(f'{i:02}', x) for i, x in enumerate(models, 1)])
blocks = re.split(r'^## M\d{2}: .+$', m, flags=re.M)[1:]
field_total = 0
for i, block in enumerate(blocks, 1):
    rows = table_rows(block.split('## Selection', 1)[0], r'^\| (?:' + '|'.join(map(re.escape, fields)) + r') \|')
    field_total += len(rows)
    check(f'M{i:02}: ten substantive fields', [v[0] for v in rows] == fields and all(len(v) == 2 and len(v[1]) >= 60 for v in rows))
check('130 required model fields', field_total == 130)
check('17 source records with resolvable local anchors',
      re.findall(r'<a id="(r\d{2})">', r) == [f'r{i:02}' for i in range(1, 18)] and
      len(re.findall(r'^### R\d{2}:', r, re.M)) == 17)
check('source access date and limitations explicit', all(x in r for x in [
    '10 September 2026', 'publisher-indexed abstract', 'HTTP 503',
    'not provider selection', 'No affiliate enrolment', 'not a calibrated price']))
check('all synthetic fixtures present', re.findall(r'^### (S\d{2}):', r, re.M) == [f'S{i:02}' for i in range(1, 9)])
check('16 failure and repair rows', len(table_rows(a, r'^\| F\d{2} \|')) == 16)
check('no automatic ingredient/score rule', all(x in a for x in [
    'not a mandatory sales device', 'Do not average these into an offer score',
    'No universal target margin', 'guarantees and urgency can be absent']))
check('customer evidence and approved versions preserved', all(x in a for x in [
    'CR01–CR06', 'Keep the current accepted offer', 'supporting/challenging evidence',
    'retain unaffected accepted choices']))
check('overlapping labels cannot duplicate transactions', all(x in m for x in [
    'not thirteen mutually exclusive', 'Record each transaction once',
    'Do not add overlapping', 'reconcile event identities']))
check('cash/recognition/tax boundary explicit', all(x in a+m for x in [
    'not automatically accounting revenue', 'Taxes, total customer obligation',
    'conditional commission and collected cash', 'Actual recognition, tax, principal/agent']))
check('current stage does not authorise implementation', all(x in a for x in [
    'No live prices', 'Stage 1 human/professional boundary', 'Stage 8 receives uncertain claims']))

# Local document links can target the inspected remote tree when there is
# no full checkout. The manifest contains only real returned path names.
manifest = root / 'base-tree-paths.json'
known = set(json.loads(manifest.read_text())) if manifest.exists() else set()
links = re.findall(r'\]\(([^)#]+\.md)(?:#[^)]*)?\)', a+m+r)
check('all relative documentation links resolve', all(
    (logs / p).exists() or 'docs/research-logs/' + p in known for p in links))

# Synthetic calculations. Rates and quantities are fixture inputs, never
# current provider prices, empirical effects or recommended business goals.
def economics(quantity, list_price, discount, refund_rate, unit_cost, fee_rate):
    q, p, d, f, k, fee = map(lambda x: D(str(x)),
                            [quantity, list_price, discount, refund_rate, unit_cost, fee_rate])
    gross = q*p
    charged = gross*(1-d)
    refunds = charged*f
    net = charged-refunds
    costs = q*k+charged*fee
    return gross, gross-charged, refunds, net, costs, net-costs

base = economics(50, 100, 0, '.05', 60, '.03')
alt = economics(60, 100, '.20', '.05', 60, '.03')
check('S01 baseline economics', base == tuple(map(D, ['5000','0','250','4750','3150','1600'])), base)
check('S01 discounted economics', alt == tuple(map(D, ['6000','1200','240','4560','3744','816'])), alt)
check('S01 acquisition-adjusted contribution', (base[-1]-900, alt[-1]-900) == (D(700), D(-84)), (base[-1]-900, alt[-1]-900))

check('S01 capacity cannot restore baseline', required == 118 and required > 80, required)

def cash_path(deposit):
    balance = D(20000)
    balances = [balance]
    for amount in [D(100000)*deposit, -D(65000), D(100000)*(1-deposit)]:
        balance += amount
        balances.append(balance)
    return min(balances), balance
cash = [cash_path(d) for d in [D(0), D('.4'), D('.5')]]

q = D(150)
volume = q*D('.60')

check('S04 distinct band calculations', (volume,graduated) == (D(90),D(130)), (volume,graduated))
check('S04 band contribution and cliff', (volume-q*D('.7'),graduated-q*D('.7'),D(101)*D('.6')) == (D(-15),D(25),D('60.6')))
free = [D(100)*10-(D(100)*3+D(n)*D('.5')) for n in [900,1900]]
check('S05 free-population contribution', free == [D(250),D(-250)], free)
check('S05 stated fixed-cost residual', [x-200 for x in free] == [D(50),D(-450)])
events = {'T1': D(20), 'T2': D(35), 'T3': D(40)}
check('S06 canonical event total and expansion allocation', sum(events.values()) == 95 and D(20)+15 == events['T2'], sum(events.values()))
gmv = D(10000)
platform = gmv*D('.12')

cost = D(300)+100+200
check('S07 own fee and seller funds reconcile', (platform,seller,platform-cost) == (D(1200),D(8800),D(600)) and seller+platform == gmv)

def margin(net, cost):
    return (net-cost)/net if net > 0 else None
check('zero/negative net has no misleading margin percentage', margin(D(0),D(1)) is None and margin(D(-1),D(1)) is None)
check('markup and margin denominators differ', (D(125)-100)/100 == D('.25') and margin(D(125),D(100)) == D('.2'))
expected_tokens = ['CU−84', '118 whole orders', 'CU−45,000', 'CU−5,000', 'CU5,000',
                   'CU−12,600', 'CU7,400', 'CU60.60', 'CU−450', 'CU95', 'CU600']

for name, ok, detail in checks:

print(f'TOTAL {len(checks)} | PASS {passed} | FAIL {len(checks)-passed}')
for name in ['offer-and-pricing-architecture', 'money-model-taxonomy']:

    print('GIT_BLOB | '+p.name+' | '+hashlib.sha1(b'blob '+str(len(b)).encode()+b'\0'+b).hexdigest())
