## 5. Reproducible checker

Save the following code to a temporary file and run `python verify_stage04.py /path/to/repository`. In a complete checkout, documentation links resolve against actual files. In an API-only staging workspace, an optional `base-tree-paths.json` supplies the exact already-inspected remote path list; it does not supply or fabricate source contents. The checker's only outputs are verification text and an exit status.

```python
#!/usr/bin/env python3
"""Stage 4 document checks, not a customer-validation or product benchmark."""
from pathlib import Path
import hashlib
import json
import re
import sys

root = Path(sys.argv[1] if len(sys.argv) > 1 else '.')
logs = root / 'docs/research-logs'
name = '2026-09-10-stage-04-business-customer-and-value-model.md'
text = (logs / name).read_text(encoding='utf-8')
results = []

def check(label, condition):
    results.append({'check': label, 'result': 'PASS' if condition else 'FAIL'})

def section(start, end):
    return text.split(start, 1)[1].split(end, 1)[0]

def rows(body, prefix):
    return re.findall(r'^\| (' + prefix + r'\d{2}) \| ([^|]+) \|', body, re.M)

# These enumerations are copied from original bootstrap §11, not inferred
# from the draft. The candidate map is retained by an explicit design choice.
concerns = ['target customer', 'problem / desired progress', 'alternative / status quo',
            'value proposition', 'offer', 'acquisition channel', 'sales / conversion path',
            'price / revenue mechanism', 'delivery mechanism', 'cost structure',
            'retention / repeat / expansion', 'referral / advocacy',
            'operational constraints', 'cash timing', 'key assumptions']
roles = ['declared target customer', 'observed customer', 'buyer', 'user',
         'economic decision-maker', 'influencer']
evidence = ['interviews', 'sales calls', 'support issues', 'search / demand evidence',
            'usage data', 'conversion data', 'retention data',
            'competitive alternatives', 'willingness-to-pay evidence']
values = ['functional value', 'economic value', 'emotional value', 'social/status value',
          'risk reduction', 'time reduction', 'effort reduction', 'uncertainty reduction']
for label, body, prefix, expected in [
    ('15 selected business concerns', section('### 2.1', '## 3.'), 'B', concerns),
    ('6 customer distinctions', section('## 3.', '## 4.'), 'CR', roles),
    ('9 evidence classes', section('### 4.2', '## 5.'), 'E', evidence),
    ('8 optional value dimensions', section('## 5.', '## 6.'), 'V', values),
]:
    actual = rows(body, prefix)
    check(label, actual == [(f'{prefix}{i:02}', v) for i, v in enumerate(expected, 1)])
    table_lines = [line for line in body.splitlines() if re.match(r'^\| '+prefix+r'\d{2} \|', line)]
    check(label + ': substantive cells', all(all(len(c.strip()) >= 20 for c in line.split('|')[3:-1]) for line in table_lines))

check('explicit unknown and applicability states', all(s in text for s in [
    '**known observation**', '**hypothesis**', '**mixed evidence**', '**unknown**', '**not applicable with reason**']))
check('no universal ontology or value score', all(s in text for s in [
    'rather than a new graph service', 'do not add them into one value score', 'No universal sample size']))
check('populations and roles distinguished', 'not six mutually exclusive personas' in text and 'first two distinctions' in text)
check('conditional payer and coincident roles', '**payer** separately' in text and 'can coincide' in text)
check('prospects not silently made customers', 'Label interviewed prospects, visitors and non-buyers separately' in text)
check('evidence provenance and access fields', all(s in text for s in [
    'source/locator and collection date', 'population/role/segment', 'method/exposure or sample basis',
    'measure and denominator', 'limitations/counter-evidence', 'permitted use/access restrictions']))
check('claim-specific evidence and synthetic separation', all(s in text for s in [
    '**support**', '**challenge**', '**leave unresolved**', 'can never be promoted into observed customer evidence']))
check('observation metric interpretation confidence decision separate',
      'Keep raw observation, metric, interpretation, confidence and decision separate' in text)
check('payment terms and limited inference', 'hypothetical statement, quote acceptance, deposit, completed payment or refund' in text)
check('delivered value distinguished from promise', 'desired, perceived/promised and actually delivered value' in text)
check('decision links and bounded approval', all(s in text for s in [
    'supporting/challenging evidence', 'alternatives considered', 'review trigger', 'These states do not upgrade evidence']))
impact = section('## 7.', '## 8.')
expanded = set()
for start, end in re.findall(r'B(\d{2})(?:–B(\d{2}))?', impact):
    expanded.update(range(int(start), int(end or start) + 1))
check('target change covers B02 through B15', set(range(2, 16)).issubset(expanded))
check('no silent evidence transfer or wholesale rewrite',
      'do not relabel it as proof for the new segment' in impact and 'Review is not automatic replacement' in impact)
check('greenfield existing business and two-sided use', all(s in text for s in [
    '**Greenfield:**', '**Existing business:**', '**Marketplace or partner-mediated exchange:**']))
probes = section('## 9.', '## 10.')
blocks = re.split(r'(?m)^### Probe ', probes)[1:]
check('four complete synthetic traceability probes', len(blocks) == 4 and all(
    all(s in b for s in ['**Synthetic setup:**', '**Trace:**', '**Disposition:**', '**Observed design-check result:**', 'PASS.']) for b in blocks))
probe_a_rows = re.findall(r'^\| B(\d{2}) \| (.+) \|$', blocks[0], re.M)
check('probe A exercises all 15 map fields', [n for n, _ in probe_a_rows] == [f'{i:02}' for i in range(1, 16)] and all(len(v) > 40 for _, v in probe_a_rows))
check('probes not claimed as customers or benchmarks',
      'not collected customer evidence, primary progressive examples, agent benchmarks' in probes)
check('later-stage implementation not preclaimed', all(s in text for s in [
    'Detailed offer/pricing architecture remains Stage 5', 'No such stage is marked complete', 'No current-stage user decision is unresolved']))

# Read from an actual checkout when available. In an API-only workspace the
# manifest is the exact known remote tree, not placeholder source-file content.
manifest_path = root / 'base-tree-paths.json'
known = set(json.loads(manifest_path.read_text())) if manifest_path.exists() else set()
link_targets = re.findall(r'\]\(([^)#]+\.md)(?:#[^)]*)?\)', text)
check('all model links resolve in workspace or inspected remote tree', all(
    (logs / p).is_file() or 'docs/research-logs/' + p in known for p in link_targets))
check('all generated paths remain research documentation', all(
    p.startswith('docs/research-logs/') and p.endswith('.md') for p in [
    'docs/research-logs/' + name, 'docs/research-logs/2026-09-10-stage-04-conformance.md',
    'docs/research-logs/bootstrap-2-progress.md', 'docs/research-logs/2026-09-10-bootstrap-2-progress-through-stage-03.md']))
check('original source and accepted boundary pinned',
      'a0ac1b6c78a0cde8ccd9aca6fda6c5b0880c516d' in text and '885b14b5738f39a878a299542fd870186548d8be' in text)
check('local role IDs do not collide with capability IDs', not re.search(r'^\| C\d{2} \|', text, re.M))

for result in results:
    print(result['result'] + ' | ' + result['check'])
passed = sum(r['result'] == 'PASS' for r in results)
print(f'TOTAL: {len(results)}; PASS: {passed}; FAIL: {len(results)-passed}')
print('MODEL_GIT_BLOB: ' + hashlib.sha1(b'blob ' + str(len(text.encode())).encode() + b'\0' + text.encode()).hexdigest())
sys.exit(0 if passed == len(results) else 1)

```
