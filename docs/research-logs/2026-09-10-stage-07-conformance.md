def finite(*values: D) -> None:
    if any(not value.is_finite() for value in values):
        raise ValueError('non-finite numeric input')


def nonnegative(*values: D) -> None:
    finite(*values)
    if any(value < 0 for value in values):
        raise ValueError('negative input where a nonnegative amount is required')


def ratio(numerator: D, denominator: D) -> D | None:
    finite(numerator, denominator)
    if denominator < 0:
        raise ValueError('negative denominator')
    return numerator / denominator if denominator else None


def resource(available: D, reserve: D, existing: D, shared: D,
             setup: D, unit: D, quantity: int = 0) -> dict:
    nonnegative(available, reserve, existing, shared, setup, unit)
    if type(quantity) is not int or quantity < 0:
        raise ValueError('quantity must be a nonnegative integer')
    remaining = available - reserve - existing - shared - setup
    bound = max(0, int((remaining / unit).to_integral_value(rounding=ROUND_FLOOR))) if unit else None
    return {'remaining': remaining, 'bound': bound,
            'headroom': remaining - quantity * unit}


def backlog(opening: int, incoming: int, completed: int, cancelled: int) -> int:
    if any(type(x) is not int or x < 0 for x in (opening, incoming, completed, cancelled)):
        raise ValueError('invalid work counts')
    result = opening + incoming - completed - cancelled
    if result < 0:
        raise ValueError('work conservation failed')
    return result


def cost_groups(rows: list[tuple[str, str, D]]) -> dict[str, D]:
    seen: set[str] = set()
    totals = {'K': D(0), 'A': D(0), 'H': D(0)}
    for identity, group, amount in rows:
        nonnegative(amount)
        if not identity or identity in seen or group not in totals:
            raise ValueError('duplicate/invalid cost identity or group')
        seen.add(identity)
        totals[group] += amount
    return totals


def retained_revenue(opening: D, lost: D, contraction: D, expansion: D) -> dict:
    nonnegative(opening, lost, contraction, expansion)
    if lost + contraction > opening:
        raise ValueError('overlapping/excessive losses')
    gross = opening - lost - contraction
    return {'gross_amount': gross, 'net_amount': gross + expansion,
            'grr': ratio(gross, opening), 'nrr': ratio(gross + expansion, opening)}


def finite_value(margin: D, churn: D, discount: D, periods: int) -> D:
    finite(margin)
    nonnegative(churn, discount)
    if churn > 1 or type(periods) is not int or periods < 1:
        raise ValueError('invalid probability or horizon')
    q = (1 - churn) / (1 + discount)
    return sum((margin * q**t for t in range(periods)), D(0))


def infinite_constant_value(margin: D, churn: D, discount: D) -> D | None:
    finite_value(margin, churn, discount, 1)  # same validated conventions
    return margin * (1 + discount) / (churn + discount) if churn + discount else None


def cash_calendar(opening: D, floor: D,
                  events: list[tuple[str, str, D]]) -> dict:
    finite(opening)
    nonnegative(floor)
    seen: set[str] = set()
    previous: date | None = None
    running = opening
    balances = [running]
    breach = 'opening' if running < floor else None
    for identity, iso_day, amount in events:
        finite(amount)
        day = date.fromisoformat(iso_day)
        if not identity or identity in seen or (previous is not None and day < previous):
            raise ValueError('duplicate identity or nonchronological cash event')
        seen.add(identity)
        previous = day
        running += amount
        balances.append(running)
        if breach is None and running < floor:
            breach = iso_day
    low = min(balances)
    return {'balances': balances, 'closing': running, 'minimum': low,
            'gap': max(D(0), floor - low), 'first_breach': breach}


def payback(acquisition: D, net_cash: list[D]) -> dict:
    nonnegative(acquisition)
    running = -acquisition
    balances: list[D] = []
    crossing = 0 if running >= 0 else None
    for index, amount in enumerate(net_cash, 1):
        finite(amount)
        running += amount
        balances.append(running)
        if crossing is None and running >= 0:
            crossing = index
    return {'first_crossing': crossing, 'balances': balances,
            'later_deficit': crossing is not None and any(
                x < 0 for x in balances[max(0, crossing):])}


def cycle_days(stock: D, cogs: D, receivables: D, credit_sales: D,
               payables: D, credit_purchases: D, days: D) -> dict:
    nonnegative(stock, cogs, receivables, credit_sales, payables, credit_purchases, days)
    if not days:
        raise ValueError('period must be positive')
    # Multiply before division to avoid unnecessary intermediate decimal rounding.
    values = [ratio(stock * days, cogs), ratio(receivables * days, credit_sales),
              ratio(payables * days, credit_purchases)]
    return {'dio': values[0], 'dso': values[1], 'dpo': values[2],
            'cycle': None if any(x is None for x in values) else values[0] + values[1] - values[2]}


def approximate_runway(usable: D, floor: D, burn: D) -> D | None:
    finite(usable, burn)
    nonnegative(floor)
    return max(D(0), usable - floor) / burn if burn > 0 else None


def growth_gate(states: dict[str, tuple[str, str]]) -> str:
    required = {f'GG{i:02}' for i in range(1, 8)}
    if set(states) != required:
        return 'BLOCKED'
    for key, (state, reason) in states.items():
        if state not in {'PASS', 'FAIL', 'BLOCKED', 'NOT APPLICABLE'}:
            raise ValueError('unrecognised evaluation result')
        if state == 'NOT APPLICABLE' and (key != 'GG05' or not reason.strip()):
            return 'BLOCKED'
    if any(state == 'FAIL' for state, _ in states.values()):
        return 'FAIL'
    if any(state == 'BLOCKED' for state, _ in states.values()):
        return 'BLOCKED'
    return 'PASS'  # combines supplied results; does not establish their truth or authority


def expect_error(fn) -> bool:
    try:
        fn()
    except (ValueError, TypeError):
        return True
    return False
